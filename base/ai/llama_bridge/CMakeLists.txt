cmake_minimum_required(VERSION 3.18)
project(libernt_llama_bridge LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# LiberNT uses native Win32 threads via libc++ + kernel32 (no libpthreads).
set(CMAKE_USE_WIN32_THREADS_INIT 1 CACHE INTERNAL "")
set(CMAKE_USE_PTHREADS_INIT 0 CACHE INTERNAL "")
set(CMAKE_HAVE_THREADS_LIBRARY 1 CACHE INTERNAL "")
set(CMAKE_THREAD_LIBS_INIT "" CACHE STRING "" FORCE)
set(Threads_FOUND TRUE CACHE BOOL "" FORCE)

set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_APP OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_UI OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_HTML OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)

add_subdirectory("${LLAMA_CPP_SOURCE_DIR}" llama_cpp EXCLUDE_FROM_ALL)

add_library(llama_dll SHARED llama_bridge.cpp)
set_target_properties(llama_dll PROPERTIES
    OUTPUT_NAME "llama"
    PREFIX ""
    SUFFIX ".dll")
target_link_libraries(llama_dll PRIVATE llama)
target_include_directories(llama_dll PRIVATE "${LLAMA_CPP_SOURCE_DIR}/include")
